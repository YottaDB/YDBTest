# [YDB#1288] $C()/$ZCH() arguments in a ZWRITE format string and trigger -pieces= values of           #
# 2147483648 or more are rejected, not wrapped around modulo 2**32                                    #
#----------------------------------------------------------------------------------------------------#

# The A2I() macro, which parses these numbers, kept going after an int overflow as long as the value
# stayed positive, so it returned the number modulo 2**32: $C(4294967361) was read as $C(65), and
# -pieces=4294967298 as -pieces=2. When the value went negative it stopped, and a debug build failed
# an assert, for example on $C(2147483713).
#
# Without the fix, the checks below of numbers that wrap to a valid code print WRONG, and on a debug
# build the checks of numbers that go negative, such as 2147483713, fail an assert instead.

# Stage 1 : $ZWRITE(str,1) of a $C() or $ZCH() with an argument too large for an int, and controls
# Command : $ydb_dist/yottadb -run zwr^ydb1288
#   expect "" for each argument of 2147483648 or more, and the character for each control

PASS : $zwrite("$ZCH(4294967361)",1) returned ""

PASS : $zwrite("$C(4294967296065)",1) returned ""
PASS : $zwrite("$C(4294967306)",1) returned ""
PASS : $zwrite("$C(2147483648)",1) returned ""
PASS : $zwrite("$C(2147483713)",1) returned ""
PASS : $zwrite("$C(18446744073709551681)",1) returned ""
PASS : $zwrite("$C(65,4294967361)",1) returned ""
PASS : $zwrite("""X""_$C(4294967361)",1) returned ""
PASS : $zwrite("$C(2147483647)",1) returned ""
PASS : $zwrite("$C(65)",1) returned "A"
PASS : $zwrite("$C(0065)",1) returned "A"
PASS : $zwrite("$ZCH(255)",1) returned $C(255)
PASS : $zwrite("$C(1,2,3)",1) returned $C(1,2,3)
