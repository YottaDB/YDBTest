#----------------------------------------------------------------------------------------------------#
# [YDB#1338] TRESTART after a TPQUIT or BREAK in Direct Mode, a TSTART and an argumentless NEW issues #
# TRESTNEW and TRESTLOC, and without the NEW issues TRESTNOT and TRESTLOC                            #
#----------------------------------------------------------------------------------------------------#

# A TSTART in Direct Mode uses the frame under the Direct Mode frame as its restart frame. When Direct
# Mode was entered by a BREAK in $ZTRAP, that is the $ZTRAP frame, which is not a counted frame. A
# restart cannot resume in that frame, and an argumentless NEW puts its new symbol table below it, where
# a restart cannot undo it. TRESTART now issues TRESTNEW and TRESTLOC, before it unwinds anything, if a
# TSTART is such a TSTART and an argumentless NEW followed it, and TRESTNOT and TRESTLOC, also before it
# unwinds anything, if the outermost TSTART is such a TSTART with no NEW after it.
# A QUIT with an argument that issues TPQUIT in Direct Mode leaves Direct Mode running in such a frame.
# Direct Mode prompts and blank lines are removed from the output below. Each check prints PASS or FAIL.

# Stage 1 : Direct Mode, TSTART and QUIT 1 (TPQUIT), then TSTART (d) and an argumentless NEW, then TRESTART
# Before the fix : GTMASSERT2 in tp_unwind.c (curr_symval == tp_pointer->sym)
# With the fix   : expect TPQUIT, then TRESTNEW and TRESTLOC, with $TLEVEL still 2
# Input :
tstart  quit 1
tstart (d)  new
trestart
write $select(2=$tlevel:"PASS",1:"FAIL")," : $TLEVEL=",$tlevel," after TRESTNEW",!
trollback
# Command : $gtm_dist/mumps -direct < stage1.inp
%YDB-E-TPQUIT, Cannot QUIT out of a routine with an active transaction
		At M source location +1^GTM$DMOD
%YDB-E-TRESTNEW, Cannot restart transaction because the restart cannot undo an argumentless or exclusive NEW done after its TSTART
%YDB-E-TRESTLOC, Transaction start: +1^GTM$DMOD, Transaction failure: +1^GTM$DMOD
PASS : $TLEVEL=2 after TRESTNEW

# Stage 2 : a routine divides by zero with $ZTRAP="break", then in Direct Mode TSTART (d) and an argumentless
#           NEW, then TRESTART
# Before the fix : the same result
# With the fix   : expect DIVZERO, then TRESTNEW and TRESTLOC, with $TLEVEL still 1, and TROLLBACK works
# Input :
tstart (d)  new
trestart
write $select(1=$tlevel:"PASS",1:"FAIL")," : $TLEVEL=",$tlevel," after TRESTNEW",!
trollback
write $select(0=$tlevel:"PASS",1:"FAIL")," : $TLEVEL=",$tlevel," after TROLLBACK",!
halt
# Command : $gtm_dist/mumps -run ztrap^ydb1338 < stage2.inp
%YDB-E-DIVZERO, Attempt to divide by zero
		At M source location ztrap+2^ydb1338
%YDB-E-TRESTNEW, Cannot restart transaction because the restart cannot undo an argumentless or exclusive NEW done after its TSTART
%YDB-E-TRESTLOC, Transaction start: ztrap+1^ydb1338, Transaction failure: ztrap+2^ydb1338
PASS : $TLEVEL=1 after TRESTNEW
PASS : $TLEVEL=0 after TROLLBACK
