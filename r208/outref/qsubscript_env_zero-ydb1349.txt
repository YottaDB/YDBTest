#----------------------------------------------------------------------------------------------------#
# [YDB#1349] $QSUBSCRIPT(), $QLENGTH(), $NAME() and $ZCOLLATE() parse a global name whose second     #
# environment part is 0 or a $CHAR() correctly                                                       #
#----------------------------------------------------------------------------------------------------#

# When the second part of the environment of a global name was the number 0 (for example ^[1,0]a(5))
# or a $CHAR() or $ZCHAR(), the parser of the name recorded an end offset with no start offset for it,
# so every later component got the wrong offsets. $QSUBSCRIPT() returned ^ as the name and wrong
# subscripts, $ZCOLLATE() issued DBBADNSUB, and a Debug build failed an assert in is_canonic_name.c.
# Each stage runs its checks in a new process and prints PASS or WRONG.

# Stage 1 : $QLENGTH() and $QSUBSCRIPT() of ^[1,0]a(5)
# Before the fix : the name ^ and an empty subscript (Release), or an assert failure (Debug)
# With the fix   : expect the name ^a and the subscript 5
# Command : $gtm_dist/mumps -run envzero^ydb1349
PASS : ^[1,0]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|1|^a|5

# Stage 2 : $QLENGTH() and $QSUBSCRIPT() of ^[0,0]a(5,6)
# Before the fix : the name ^ and empty subscripts (Release), or an assert failure (Debug)
# With the fix   : expect the name ^a and the subscripts 5 and 6
# Command : $gtm_dist/mumps -run envzerozero^ydb1349
PASS : ^[0,0]a(5,6) : $QLENGTH and $QSUBSCRIPT(-1 to 2) are 2|0|^a|5|6

# Stage 3 : $QLENGTH() and $QSUBSCRIPT() of ^[1,$C(65)]a(5), ^[1,$ZCH(65)]a(5) and ^[1,"x"_$C(66)]a(5)
# Before the fix : the name ^ and an empty subscript (Release), or an assert failure (Debug)
# With the fix   : expect the name ^a and the subscript 5
# Command : $gtm_dist/mumps -run envchar^ydb1349
PASS : ^[1,$C(65)]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|1|^a|5
PASS : ^[1,$ZCH(65)]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|1|^a|5
PASS : ^[1,"x"_$C(66)]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|1|^a|5

# Stage 4 : $NAME(@n,1) and $NAME(@n,0) of ^[1,0]a(5,6), and $NAME(@n,1) of ^[1,$C(65)]a(5,6)
# Before the fix : the right results in a Release build, and an assert failure in a Debug build
# With the fix   : expect the name with the environment and the first subscript (or none)
# Command : $gtm_dist/mumps -run name^ydb1349
PASS : $NAME(@"^[1,0]a(5,6)",1) is ^[1,0]a(5)
PASS : $NAME(@"^[1,0]a(5,6)",0) is ^[1,0]a
PASS : $NAME(@"^[1,$C(65)]a(5,6)",1) is ^[1,"A"]a(5)

# Stage 5 : $ZCOLLATE() and $VIEW("YGVN2GDS") of ^[1,0]a(5,6), converted back to a name
# Before the fix : DBBADNSUB (Release), or an assert failure (Debug), so the lines below were WRONG
#                  lines
# With the fix   : expect ^a(5,6) for both
# Command : $gtm_dist/mumps -run zcollate^ydb1349
PASS : $ZCOLLATE($ZCOLLATE("^[1,0]a(5,6)",0),0,1) returned ^a(5,6)
PASS : $VIEW("YGDS2GVN",$VIEW("YGVN2GDS","^[1,0]a(5,6)")) returned ^a(5,6)

# Stage 6 : $QLENGTH() and $QSUBSCRIPT() of names whose environment gave the right results before
# Before the fix : the same result (this stage is a control)
# With the fix   : expect the right results
# Command : $gtm_dist/mumps -run unchanged^ydb1349
PASS : ^[1,2]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|1|^a|5
PASS : ^["0","0"]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|0|^a|5
PASS : ^[0]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|0|^a|5
PASS : ^|0|a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|0|^a|5
PASS : ^[0,1]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|0|^a|5
PASS : ^a(0,0,"0") : $QLENGTH and $QSUBSCRIPT(-1 to 3) are 3||^a|0|0|0
PASS : ^a(1,$C(65)) : $QLENGTH and $QSUBSCRIPT(-1 to 2) are 2||^a|1|A
PASS : ^a("x"_$C(66)) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1||^a|xB
PASS : ^[$C(65),2]a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|A|^a|5
PASS : ^|"x"_$C(66)|a(5) : $QLENGTH and $QSUBSCRIPT(-1 to 1) are 1|xB|^a|5

# Done
