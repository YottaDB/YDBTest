## 2. MUPIP REORG runs against a database with search indexes enabled
# REORG does not use search indexes : it rewrites the very blocks it walks, so an index it built
# would be invalidated by its own next update before the samples could be reused. What is checked
# here is that REORG completes and leaves the data intact with the feature on.
#
# The stage first proves the feature really IS on for an ordinary process, by peeking this
# process own sgmnt_addrs.search_idx_base, which is the field a search tests. Without that,
# matching answers would prove nothing : a database with the feature off matches them too.
#
# REORG own exclusion is still NOT tested, and cannot be from outside : mupip_reorg clears
# that field in REORG own address space, which is private, so no other process can read it.

## 3. A statsDB gets NO search index
# Not because its tree is small, it carries one record per live process, but because next to
# nothing searches it : a process finds its own record once and holds on to it. An index there
# would be shared memory spent for nothing, once per base region.
# A STATSDB region is named for its base region in lower case, so DEFAULT gives default, and
# statistics sharing has to be on for a process to reach it.
# mumps -run %XCMD, reading sgmnt_data.search_idx_size and search_idx_slots
# through PEEKBYNAME for region default
statsdb size  = 0
statsdb slots = 0

statsDB carries no search index, as expected
# against the base region, which has them
# mumps -run %XCMD, reading sgmnt_data.search_idx_size and search_idx_slots
# through PEEKBYNAME for region DEFAULT
base    size  = 1024
base    slots = 1024

base region carries one, as expected
