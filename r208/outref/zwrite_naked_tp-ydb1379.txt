# Each stage creates a database in its own directory, runs the lines shown in Direct Mode in a new
# process, and checks that the output has no fatal error, has the lines expected, and that the last
# line of input ran.

# Stage 1 : ZWRITE of a naked reference in a transaction, after a SET before the transaction
# Command : $gtm_dist/mumps -direct < stage1.inp, where stage1.inp has these lines
#   set ^d(1)=1,^d(2)=2
#   tstart ()
#   zwrite ^(2)
#   tcommit
#   write "end",!
# Before the fix : a Release build terminated with a SIG-11 in gvcst_search(), and a Debug build
#                  failed an assert in gvcst_data.c
# With the fix   : expect the lines ^d(2)=2
PASS : no fatal error in stage1.out
PASS : stage1.out has ^d(2)=2
PASS : the last line of input ran

# Stage 2 : the same with ^(*)
# Command : $gtm_dist/mumps -direct < stage2.inp, where stage2.inp has these lines
#   set ^d(1)=1,^d(2)=2
#   tstart ()
#   zwrite ^(*)

#   write "end",!
# Before the fix : a Release build terminated with a SIG-11 in gvcst_search(), and a Debug build
#                  failed an assert in gvcst_data.c
# With the fix   : expect the lines ^d(1)=1 ^d(2)=2
PASS : no fatal error in stage2.out
PASS : stage2.out has ^d(1)=1
PASS : stage2.out has ^d(2)=2
PASS : the last line of input ran
