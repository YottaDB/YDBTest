# Release note (from http://tinco.pair.com/bhaskar/gtm/doc/articles/GTM_V7.1-004_Release_Notes.html#GTM-DE197637833)
# 
# $STORAGE reports the correct value when there is a restriction on the data segment limit (RLIMIT_DATA) of a process. Previously, $STORAGE inappropriately returned 2GiB in such a case. (GTM-9833)
# 

# $STORAGE is the data segment size limit (RLIMIT_DATA) minus the memory YottaDB has allocated,
# which $ZREALSTOR reports, so $STORAGE+$ZREALSTOR should equal the limit in bytes.

# Case 1: Set the data segment size limit (RLIMIT_DATA) to 16384 KiB
# Run [$gtm_dist/mumps -run %XCMD 'write "$STORAGE+$ZREALSTOR=",$storage+$zrealstor,!']
# Expect 16777216. Previously, 2147483647.
$STORAGE+$ZREALSTOR=16777216

# Case 2: Set the data segment size limit (RLIMIT_DATA) to 32768 KiB
# Run [$gtm_dist/mumps -run %XCMD 'write "$STORAGE+$ZREALSTOR=",$storage+$zrealstor,!']
# Expect 33554432. Previously, 2147483647.
$STORAGE+$ZREALSTOR=33554432

# Case 3: Set the data segment size limit (RLIMIT_DATA) to 65536 KiB
# Run [$gtm_dist/mumps -run %XCMD 'write "$STORAGE+$ZREALSTOR=",$storage+$zrealstor,!']
# Expect 67108864. Previously, 2147483647.
$STORAGE+$ZREALSTOR=67108864

# Case 4: Set the data segment size limit (RLIMIT_DATA) to 131072 KiB
# Run [$gtm_dist/mumps -run %XCMD 'write "$STORAGE+$ZREALSTOR=",$storage+$zrealstor,!']
# Expect 134217728. Previously, 2147483647.
$STORAGE+$ZREALSTOR=134217728
