# ********************************************************************************************
# GTM-10603 - Test the following release note
# ********************************************************************************************
# 
# Release note (from http://tinco.pair.com/bhaskar/gtm/doc/articles/GTM_V7.1-004_Release_Notes.html#GTM-DE1959410603)
# 
# $ZMAXTPTIME and its initializing environment variable, $gtm_zmaxtptime, maintain precision to milliseconds (three digits after the decimal point). Also, both ignore values exceeding a maximum of 60 seconds. Previously $ZMAXTPTIME only provided whole seconds, and, while it limited the initial value to 60 seconds, it did not limit the SET value. (GTM-10603)
# 

### Test 1: $ZMAXTPTIME keeps millisecond precision
## SET $ZMAXTPTIME
# SET $ZMAXTPTIME=.001. Expect .001. Previously, whole seconds only.
$ZMAXTPTIME=.001
# SET $ZMAXTPTIME=.333. Expect .333. Previously, whole seconds only.
$ZMAXTPTIME=.333
# SET $ZMAXTPTIME=59.999. Expect 59.999. Previously, whole seconds only.
$ZMAXTPTIME=59.999
## Set gtm_zmaxtptime or ydb_maxtptime before process startup
# Set the environment variable to .001. Expect .001. Previously, whole seconds only.
$ZMAXTPTIME=.001
# Set the environment variable to .333. Expect .333. Previously, whole seconds only.
$ZMAXTPTIME=.333
# Set the environment variable to 59.999. Expect 59.999. Previously, whole seconds only.
$ZMAXTPTIME=59.999

### Test 2: $ZMAXTPTIME ignores values outside 0 to 60 seconds
## SET $ZMAXTPTIME
# SET $ZMAXTPTIME=.5. Expect .5.
$ZMAXTPTIME=.5
# SET $ZMAXTPTIME=61. Expect .5. Previously, 61.
$ZMAXTPTIME=.5
# SET $ZMAXTPTIME=60.001. Expect .5.
$ZMAXTPTIME=.5
# SET $ZMAXTPTIME=-1. Expect .5. Previously, 0.
$ZMAXTPTIME=.5
# SET $ZMAXTPTIME=60. Expect 60.
$ZMAXTPTIME=60
## Set gtm_zmaxtptime or ydb_maxtptime before process startup
# Set the environment variable to 61. Expect 0, the default.
$ZMAXTPTIME=0
# Set the environment variable to 60.001. Expect 0, the default.
$ZMAXTPTIME=0
# Set the environment variable to 60. Expect 60.
$ZMAXTPTIME=60

### Test 3: A fractional $ZMAXTPTIME times out a transaction
# SET $ZMAXTPTIME=.5 and hang for .1 seconds. Expect the transaction to commit.
Transaction committed
# SET $ZMAXTPTIME=.5 and hang for 2 seconds. Expect a TPTIMEOUT error. Previously, the .5 was dropped,
# which left no timeout, so the transaction committed.
TPTIMEOUT error
