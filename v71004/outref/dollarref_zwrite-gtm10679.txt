# GTM-10679 - Test the following release note
# ********************************************************************************************

# Release note (from http://tinco.pair.com/bhaskar/gtm/doc/articles/GTM_V7.1-004_Release_Notes.html#GTM-DE1959410679)
# 
# ZWRITE of a gvn maintains $REFERENCE to the last node it successfully processed. If you use ZWRITE as a debugging tool, or unlikely, in application code, in a circumstance where preserving $REFERENCE has importance, note $REFERENCE and restore it, as you would for any other command or function that references a global. Previously, ZWRITE restored the $REFERENCE to the value it had prior to the ZWRITE. (GTM-10679)
# 

### Test that ZWRITE updates $REFERENCE after a variety of other operations update $REFERENCE
## Test 1: $REFERENCE updated by SET
# set ^x=1
# set ^y=2
# set ^z=3
# zwrite ^x
^x=1
# Check $REFERENCE: Expect ^x, previously ^z.
$REFERENCE=^x

## Test 2: $REFERENCE updated by $GET
# $GET(^y)=2
# zwrite ^x
^x=1
# Check $REFERENCE: Expect ^x, previously ^y.
$REFERENCE=^x

## Test 3: $REFERENCE updated by postconditional
# write:^z
# zwrite ^x
^x=1
# Check $REFERENCE: Expect ^x, previously ^z.
$REFERENCE=^x

## Test 4: $REFERENCE updated by OPEN
# set ^socket="socket"
# open ^socket::"SOCKET"
# zwrite ^x
^x=1
# Check $REFERENCE: Expect ^x, previously ^socket.
$REFERENCE=^x

## Test 5: $REFERENCE updated by USE
# use ^socket
# zwrite ^x
^x=1
# Check $REFERENCE: Expect ^x, previously ^socket.
$REFERENCE=^x

## Test 6: $REFERENCE updated by CLOSE
# close ^socket
# zwrite ^x
^x=1
# Check $REFERENCE: Expect ^x, previously ^socket.
$REFERENCE=^x

## Test 7: $REFERENCE updated by $DATA
# if $DATA(^y)
# zwrite ^x
^x=1
# Check $REFERENCE: Expect ^x, previously ^y.
$REFERENCE=^x

## Test 8: Restore $REFERENCE from variable
# if ^y
# set restore=$REFERENCE
# zwrite ^x
^x=1
# if $data(@restore)
# Check $REFERENCE: Expect ^y, restored after ZWRITE left it at ^x.
$REFERENCE=^y

## Test 9: $REFERENCE is the last node ZWRITE displays
# set ^a(1)=1,^a(2)=2,^a(3)=3
# if ^y
# zwrite ^a
^a(1)=1
^a(2)=2
^a(3)=3
# Check $REFERENCE: Expect ^a(3), previously ^y.
$REFERENCE=^a(3)

## Test 10: A naked reference after ZWRITE uses the last node ZWRITE displays
# if ^y
# zwrite ^a
^a(1)=1
^a(2)=2
^a(3)=3
# Check ^(1): Expect 1 from ^a(1), previously a GVNAKED error, since ^y has no subscripts.
^(1)=1

## Test 11: $REFERENCE is the last node of a ZWRITE subscript range
# if ^y
# zwrite ^a(1:2)
^a(1)=1
^a(2)=2
# Check $REFERENCE: Expect ^a(2), previously ^y.
$REFERENCE=^a(2)

## Test 12: $REFERENCE is unchanged when ZWRITE displays no nodes
# if ^y
# zwrite ^a(5:9)
# Check $REFERENCE: Expect ^y.
$REFERENCE=^y
